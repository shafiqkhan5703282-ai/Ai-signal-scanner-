let prices = []; let autoTimer = null; let countdownTimer = null; let history = []; const $ = id => document.getElementById(id); function random(min, max) { return Math.random() * (max - min) + min; } function createMarket() { prices = []; let price = 100; for (let i = 0; i < 250; i++) { price += random(-0.35, 0.35); prices.push(price); } } function updateMarket() { if (prices.length === 0) { createMarket(); return; } let last = prices[prices.length - 1]; last += random(-0.35, 0.35); prices.push(last); if (prices.length > 250) { prices.shift(); } } function EMA(data, period) { if (data.length < period) { return data[data.length - 1]; } const multiplier = 2 / (period + 1); let value = data[0]; for (let i = 1; i < data.length; i++) { value = (data[i] * multiplier) + (value * (1 - multiplier)); } return value; } function RSI(data, period = 14) { if (data.length < period + 1) { return 50; } let gains = 0; let losses = 0; for ( let i = data.length - period; i < data.length; i++ ) { const change = data[i] - data[i - 1]; if (change > 0) { gains += change; } else { losses += Math.abs(change); } } if (losses === 0) { return 100; } const rs = gains / losses; return 100 - (100 / (1 + rs)); } function MACD(data) { const fast = EMA(data, 12); const slow = EMA(data, 26); return fast - slow; } function volatility(data, period = 20) { if (data.length < period + 1) { return 0; } let changes = []; for ( let i = data.length - period; i < data.length; i++ ) { changes.push( Math.abs(data[i] - data[i - 1]) ); } return changes.reduce( (a, b) => a + b, 0 ) / changes.length; } function analyze() { updateMarket(); const current = prices[prices.length - 1]; const previous = prices[prices.length - 2]; const e9 = EMA(prices, 9); const e21 = EMA(prices, 21); const e50 = EMA(prices, 50); const e100 = EMA(prices, 100); const rsi = RSI(prices, 14); const macd = MACD(prices); const momentum = current - prices[prices.length - 6]; const vol = volatility(prices); let up = 0; let down = 0; /* TREND CONFIRMATION */ if (current > e9) { up += 10; } else { down += 10; } if (e9 > e21) { up += 15; } else { down += 15; } if (e21 > e50) { up += 15; } else { down += 15; } if (e50 > e100) { up += 10; } else { down += 10; } /* RSI CONFIRMATION */ if (rsi >= 52 && rsi <= 67) { up += 15; } if (rsi <= 48 && rsi >= 33) { down += 15; } /* MACD CONFIRMATION */ if (macd > 0) { up += 15; } if (macd < 0) { down += 15; } /* MOMENTUM */ if (momentum > 0) { up += 10; } if (momentum < 0) { down += 10; } /* CANDLE PRESSURE */ if (current > previous) { up += 10; } if (current < previous) { down += 10; } /* VOLATILITY FILTER Extremely weak movement = WAIT instead of forcing a trade. */ let signal = "WAIT"; const difference = Math.abs(up - down); let confidence = Math.round( Math.min( 99, Math.max(up, down) ) ); /* STRONG CONFIRMATION */ if ( up > down && difference >= 30 && confidence >= 75 && rsi >= 50 && rsi < 70 ) { signal = "UP"; } else if ( down > up && difference >= 30 && confidence >= 75 && rsi <= 50 && rsi > 30 ) { signal = "DOWN"; } /* TREND */ let trend = "SIDEWAYS"; if ( e9 > e21 && e21 > e50 && e50 > e100 ) { trend = "UPTREND"; } else if ( e9 < e21 && e21 < e50 && e50 < e100 ) { trend = "DOWNTREND"; } /* MOMENTUM TEXT */ let momentumText = "NEUTRAL"; if (momentum > 0) { momentumText = "BULLISH"; } else if (momentum < 0) { momentumText = "BEARISH"; } /* VERY LOW VOLATILITY FILTER */ if (vol < 0.03) { signal = "WAIT"; } return { signal: signal, confidence: confidence, rsi: rsi, trend: trend, momentum: momentumText }; } function display(result) { const orb = $("orb"); const signal = $("signal"); orb.className = "orb"; if (result.signal === "UP") { orb.classList.add("up"); orb.textContent = "UP"; signal.textContent = "CALL / UP"; signal.className = "signalText green"; } else if (result.signal === "DOWN") { orb.classList.add("down"); orb.textContent = "DOWN"; signal.textContent = "PUT / DOWN"; signal.className = "signalText red"; } else { orb.classList.add("wait"); orb.textContent = "WAIT"; signal.textContent = "NO TRADE"; signal.className = "signalText yellow"; } $("confidence").textContent = result.confidence + "%"; $("rsi").textContent = result.rsi.toFixed(1); $("trend").textContent = result.trend; $("momentum").textContent = result.momentum; if (result.signal === "WAIT") { $("details").textContent = "Market confirmation is not strong enough."; } else { $("details").textContent = "Multiple indicators confirmed the direction."; } } function generate() { const result = analyze(); display(result); if (result.signal !== "WAIT") { history.unshift({ pair: $("pair").value, signal: result.signal, confidence: result.confidence, time: new Date().toLocaleTimeString() }); if (history.length > 50) { history.pop(); } renderHistory(); } return result; } function renderHistory() { $("history").innerHTML = ""; history.forEach(item => { const row = document.createElement("div"); row.className = "historyRow"; row.innerHTML = ` ${item.pair} ${item.signal} ${item.confidence}% ${item.time} `; $("history") .appendChild(row); }); } function startAuto() { if (autoTimer) { return; } generate(); let seconds = 5; $("countdown").textContent = "Next signal in " + seconds + "s"; countdownTimer = setInterval(() => { seconds--; if (seconds <= 0) { seconds = 5; } $("countdown").textContent = "Next signal in " + seconds + "s"; }, 1000); autoTimer = setInterval(() => { generate(); }, 5000); } function stopAuto() { clearInterval(autoTimer); clearInterval(countdownTimer); autoTimer = null; countdownTimer = null; $("countdown").textContent = "Stopped"; } $("scan").addEventListener( "click", generate ); $("auto").addEventListener( "click", startAuto ); $("stop").addEventListener( "click", stopAuto ); createMarket();
