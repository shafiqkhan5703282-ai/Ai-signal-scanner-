let prices = []; let autoTimer = null; let countdownTimer = null; let history = []; const $ = id => document.getElementById(id); function random(min, max) { return Math.random() * (max - min) + min; } function createMarket() { prices = []; let p = 100; for (let i = 0; i < 200; i++) { p += random(-0.45, 0.45); prices.push(p); } } function updateMarket() { if (!prices.length) { createMarket(); return; } let last = prices[prices.length - 1]; // Simulated market movement last += random(-0.45, 0.45); prices.push(last); if (prices.length > 200) { prices.shift(); } } function EMA(data, period) { if (data.length < period) { return data[data.length - 1]; } const k = 2 / (period + 1); let value = data[0]; for (let i = 1; i < data.length; i++) { value = data[i] * k + value * (1 - k); } return value; } function RSI(data, period = 14) { if (data.length < period + 1) { return 50; } let gain = 0; let loss = 0; for (let i = data.length - period; i < data.length; i++) { const change = data[i] - data[i - 1]; if (change > 0) { gain += change; } else { loss += Math.abs(change); } } if (loss === 0) return 100; const rs = gain / loss; return 100 - (100 / (1 + rs)); } function MACD(data) { return EMA(data, 12) - EMA(data, 26); } function analyze() { updateMarket(); const current = prices[prices.length - 1]; const e9 = EMA(prices, 9); const e21 = EMA(prices, 21); const e50 = EMA(prices, 50); const rsi = RSI(prices); const macd = MACD(prices); const momentum = current - prices[prices.length - 6]; const previous = prices[prices.length - 2]; let upScore = 0; let downScore = 0; // EMA 9 if (current > e9) { upScore += 15; } else { downScore += 15; } // EMA 9 / 21 if (e9 > e21) { upScore += 15; } else { downScore += 15; } // EMA 21 / 50 if (e21 > e50) { upScore += 15; } else { downScore += 15; } // RSI if (rsi >= 52 && rsi <= 68) { upScore += 15; } if (rsi <= 48 && rsi >= 32) { downScore += 15; } // MACD if (macd > 0) { upScore += 15; } else { downScore += 15; } // Momentum if (momentum > 0) { upScore += 15; } if (momentum < 0) { downScore += 15; } // Current candle pressure if (current > previous) { upScore += 10; } else if (current < previous) { downScore += 10; } const difference = Math.abs(upScore - downScore); const strongest = Math.max(upScore, downScore); let confidence = Math.round(Math.min(99, strongest)); let signal = "WAIT"; // High-selectivity confirmation if ( upScore > downScore && difference >= 25 && confidence >= 75 ) { signal = "UP"; } else if ( downScore > upScore && difference >= 25 && confidence >= 75 ) { signal = "DOWN"; } // Trend let trend = "SIDEWAYS"; if (e9 > e21 && e21 > e50) { trend = "UPTREND"; } else if (e9 < e21 && e21 < e50) { trend = "DOWNTREND"; } // Momentum let momentumText = "NEUTRAL"; if (momentum > 0) { momentumText = "BULLISH"; } else if (momentum < 0) { momentumText = "BEARISH"; } return { signal, confidence, rsi, trend, momentum: momentumText }; } function display(result) { const orb = $("orb"); const signal = $("signal"); orb.className = "orb"; if (result.signal === "UP") { orb.classList.add("up"); orb.textContent = "UP"; signal.textContent = "CALL / UP"; signal.className = "signalText green"; } else if (result.signal === "DOWN") { orb.classList.add("down"); orb.textContent = "DOWN"; signal.textContent = "PUT / DOWN"; signal.className = "signalText red"; } else { orb.classList.add("wait"); orb.textContent = "WAIT"; signal.textContent = "NO TRADE"; signal.className = "signalText yellow"; } $("confidence").textContent = result.confidence + "%"; $("rsi").textContent = result.rsi.toFixed(1); $("trend").textContent = result.trend; $("momentum").textContent = result.momentum; $("details").textContent = result.signal === "WAIT" ? "Confirmation is not strong enough. WAIT." : "Multiple confirmations detected."; } function generate() { const result = analyze(); display(result); if (result.signal !== "WAIT") { history.unshift({ pair: $("pair").value, signal: result.signal, confidence: result.confidence, time: new Date().toLocaleTimeString() }); if (history.length > 50) { history.pop(); } renderHistory(); } return result; } function renderHistory() { $("history").innerHTML = ""; history.forEach(item => { const row = document.createElement("div"); row.className = "historyRow"; row.innerHTML = ` ${item.pair} ${item.signal} ${item.confidence}% ${item.time} `; $("history").appendChild(row); }); } function startAuto() { if (autoTimer) return; generate(); let seconds = 5; $("countdown").textContent = "Next signal in " + seconds + "s"; countdownTimer = setInterval(() => { seconds--; if (seconds <= 0) { seconds = 5; } $("countdown").textContent = "Next signal in " + seconds + "s"; }, 1000); autoTimer = setInterval(() => { generate(); }, 5000); } function stopAuto() { clearInterval(autoTimer); clearInterval(countdownTimer); autoTimer = null; countdownTimer = null; $("countdown").textContent = "Stopped"; } $("scan").addEventListener( "click", generate ); $("auto").addEventListener( "click", startAuto ); $("stop").addEventListener( "click", stopAuto ); createMarket();
